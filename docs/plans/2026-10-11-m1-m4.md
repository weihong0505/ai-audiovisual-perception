## 评审关注点（Review Focus）

1. **中文关键词检索**：trigram 要求查询词 ≥3 字符，否则返回空；`search_transcripts` 对 <3 字符查询回退 `LIKE '%q%'`——否则搜「货车」有结果、搜「门铃」误为空。→ 测试：`tests/test_search.py::test_search_chinese_3chars`、`::test_search_short_query_like_fallback`（Task 7）。
2. **空文本/空总结不落库**：ASR 或 VLM 返回空白时 `insert_transcription` / `insert_clip_analysis` 返回 `None`，不产生脏行。→ 测试：`tests/test_db.py::test_insert_transcription_skips_empty`（Task 2）。
3. **关键帧必须挂到真实视频段**：帧在视频段关闭（拿到 `video_segments.id`）之后才落库，避免 `frames.video_segment_id` 悬空或指向错误段。→ 测试：`tests/test_recorder.py` 断言 `frames.video_segment_id == video_segments.id`（Task 5）。
4. **时间格式无日界回绕**：`_fmt_ms` 纯毫秒换算，跨 24h 正确（25:05:00 而非 01:05:00）。→ 测试：`tests/test_ask.py::test_fmt_ms_no_day_rollover`（Task 8）。
5. **Ollama 不可用必须显式报错**：连接失败/HTTP 错误抛异常向上传播，不静默返回空或编造。→ 测试：`tests/test_ask.py::test_ask_ollama_raises_on_connection_error`（Task 8）。

```python
    with open(path, "r", encoding="utf-8") as f:
        user = yaml.safe_load(f) or {}
    cfg = {}
    for k, v in DEFAULTS.items():
        cfg[k] = {**v, **(user.get(k) or {})}
    if "cameras" in user:
        cfg["cameras"] = user["cameras"]
    return cfg
```

```python
"""SQLite 存储层：建表、写入、FTS5(trigram) 手动同步。"""
import sqlite3
from pathlib import Path

SCHEMA = """
CREATE TABLE IF NOT EXISTS cameras (
  id INTEGER PRIMARY KEY, name TEXT NOT NULL, source TEXT NOT NULL,
  created_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now'))
);
CREATE TABLE IF NOT EXISTS video_segments (
  id INTEGER PRIMARY KEY, camera_id INTEGER NOT NULL REFERENCES cameras(id),
  start_ms INTEGER NOT NULL, end_ms INTEGER NOT NULL, file_path TEXT NOT NULL,
  created_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now'))
);
CREATE INDEX IF NOT EXISTS idx_seg_cam_ts ON video_segments(camera_id, start_ms);
CREATE TABLE IF NOT EXISTS transcriptions (
  id INTEGER PRIMARY KEY, camera_id INTEGER,
  start_ms INTEGER NOT NULL, end_ms INTEGER NOT NULL, text TEXT NOT NULL,
  device TEXT NOT NULL DEFAULT 'mic', model TEXT NOT NULL DEFAULT 'faster-whisper',
  created_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now'))
);
CREATE INDEX IF NOT EXISTS idx_trans_start ON transcriptions(start_ms);
CREATE VIRTUAL TABLE IF NOT EXISTS transcriptions_fts USING fts5(text, tokenize='trigram');
CREATE TABLE IF NOT EXISTS frames (
  id INTEGER PRIMARY KEY, camera_id INTEGER NOT NULL REFERENCES cameras(id),
  video_segment_id INTEGER NOT NULL REFERENCES video_segments(id),
  captured_ms INTEGER NOT NULL, offset_ms INTEGER NOT NULL, file_path TEXT NOT NULL,
  created_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now'))
);
CREATE INDEX IF NOT EXISTS idx_frames_cam_ts ON frames(camera_id, captured_ms);
CREATE TABLE IF NOT EXISTS clip_analyses (
  id INTEGER PRIMARY KEY, camera_id INTEGER NOT NULL REFERENCES cameras(id),
  start_ms INTEGER NOT NULL, end_ms INTEGER NOT NULL, summary TEXT NOT NULL, model TEXT NOT NULL,
  created_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now'))
);
CREATE INDEX IF NOT EXISTS idx_clip_cam_ts ON clip_analyses(camera_id, start_ms);
CREATE VIRTUAL TABLE IF NOT EXISTS clip_analyses_fts USING fts5(summary, tokenize='trigram');
CREATE TABLE IF NOT EXISTS events (
  id INTEGER PRIMARY KEY, ts_ms INTEGER NOT NULL, kind TEXT NOT NULL, payload TEXT NOT NULL
);
"""

def connect(db_path):
    p = Path(db_path)
    p.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(str(p))
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA journal_mode=WAL")
    conn.execute("PRAGMA foreign_keys=ON")
    return conn

def init_db(db_path):
    conn = connect(db_path)
    conn.executescript(SCHEMA)
    conn.commit()
    conn.close()

def insert_camera(db_path, name, source, cam_id=None):
    conn = connect(db_path)
    if cam_id is not None:
        conn.execute("INSERT OR IGNORE INTO cameras(id, name, source) VALUES (?,?,?)",
                     (cam_id, name, str(source)))
        rid = cam_id
    else:
        cur = conn.execute("INSERT INTO cameras(name, source) VALUES (?,?)", (name, str(source)))
        rid = cur.lastrowid
    conn.commit(); conn.close()
    return rid

def insert_video_segment(db_path, camera_id, start_ms, end_ms, file_path):
    conn = connect(db_path)
    cur = conn.execute(
        "INSERT INTO video_segments(camera_id, start_ms, end_ms, file_path) VALUES (?,?,?,?)",
        (camera_id, start_ms, end_ms, file_path))
    conn.commit(); rid = cur.lastrowid; conn.close()
    return rid

def insert_transcription(db_path, camera_id, start_ms, end_ms, text, device="mic", model="faster-whisper"):
    text = (text or "").strip()
    if not text:
        return None
    conn = connect(db_path)
    cur = conn.execute(
        "INSERT INTO transcriptions(camera_id, start_ms, end_ms, text, device, model) VALUES (?,?,?,?,?,?)",
        (camera_id, start_ms, end_ms, text, device, model))
    rid = cur.lastrowid
    conn.execute("INSERT INTO transcriptions_fts(rowid, text) VALUES (?, ?)", (rid, text))
    conn.commit(); conn.close()
    return rid

def insert_frame(db_path, camera_id, video_segment_id, captured_ms, offset_ms, file_path):
    conn = connect(db_path)
    cur = conn.execute(
        "INSERT INTO frames(camera_id, video_segment_id, captured_ms, offset_ms, file_path) VALUES (?,?,?,?,?)",
        (camera_id, video_segment_id, captured_ms, offset_ms, file_path))
    conn.commit(); rid = cur.lastrowid; conn.close()
    return rid

def insert_clip_analysis(db_path, camera_id, start_ms, end_ms, summary, model):
    summary = (summary or "").strip()
    if not summary:
        return None
    conn = connect(db_path)
    cur = conn.execute(
        "INSERT INTO clip_analyses(camera_id, start_ms, end_ms, summary, model) VALUES (?,?,?,?,?)",
        (camera_id, start_ms, end_ms, summary, model))
    rid = cur.lastrowid
    conn.execute("INSERT INTO clip_analyses_fts(rowid, summary) VALUES (?, ?)", (rid, summary))
    conn.commit(); conn.close()
    return rid

def insert_event(db_path, ts_ms, kind, payload):
    conn = connect(db_path)
    cur = conn.execute("INSERT INTO events(ts_ms, kind, payload) VALUES (?,?,?)", (ts_ms, kind, payload))
    conn.commit(); rid = cur.lastrowid; conn.close()
    return rid
```

```python
"""音频源：抽象 + 文件/麦克风实现，产出 float32 16kHz 单声道 chunks。"""
import subprocess
import numpy as np

class AudioSource:
    def chunks(self):
        raise NotImplementedError

class FileAudioSource(AudioSource):
    def __init__(self, path, sample_rate=16000):
        self.path = str(path)
        self.sample_rate = sample_rate

    def chunks(self):
        cmd = ["ffmpeg", "-i", self.path, "-f", "s16le", "-ac", "1",
               "-ar", str(self.sample_rate), "pipe:1"]
        proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL)
        chunk_bytes = self.sample_rate * 2  # 1 秒 = sample_rate 采样 × 2 字节
        while True:
            raw = proc.stdout.read(chunk_bytes)
            if not raw:
                break
            yield np.frombuffer(raw, dtype=np.int16).astype(np.float32) / 32768.0
        proc.wait()

class MicAudioSource(AudioSource):
    def __init__(self, device=None, sample_rate=16000):
        self.device = device
        self.sample_rate = sample_rate

    def chunks(self):
        import sounddevice as sd
        with sd.InputStream(samplerate=self.sample_rate, channels=1,
                            device=self.device, dtype="float32") as stream:
            while True:
                data, _ = stream.read(self.sample_rate)  # 每次 1 秒
                yield data[:, 0] if data.ndim > 1 else data
```

```python
"""语音转写：faster-whisper 封装。"""
import numpy as np
from faster_whisper import WhisperModel

class Transcriber:
    def __init__(self, model_size="small", device="cpu", language="zh"):
        self.model = WhisperModel(model_size, device=device, compute_type="int8")

    def transcribe(self, audio: np.ndarray, sample_rate=16000):
        if audio is None or len(audio) == 0:
            return []
        segments, _info = self.model.transcribe(
            audio, language=None, beam_size=5, vad_filter=True)
        return [{"start": s.start, "end": s.end, "text": s.text.strip()} for s in segments]
```

```python
"""连续录制：VideoSource 抽象 + CameraRecorder（分段 mp4 + 关键帧落库）。"""
import time
from pathlib import Path
import cv2
from store.db import insert_video_segment, insert_frame

class VideoSource:
    def read(self):
        raise NotImplementedError
    def fps(self):
        raise NotImplementedError
    def release(self):
        raise NotImplementedError

class CameraVideoSource(VideoSource):
    def __init__(self, source):
        # source 可为设备号(int) 或 rtsp/rtmp URL(str)
        s = str(source)
        self.cap = cv2.VideoCapture(int(s) if s.isdigit() else s)
    def read(self):
        return self.cap.read()
    def fps(self):
        return self.cap.get(cv2.CAP_PROP_FPS) or 30
    def release(self):
        self.cap.release()

class FileVideoSource(VideoSource):
    def __init__(self, path):
        self.cap = cv2.VideoCapture(str(path))
    def read(self):
        return self.cap.read()
    def fps(self):
        return self.cap.get(cv2.CAP_PROP_FPS) or 30
    def release(self):
        self.cap.release()

class CameraRecorder:
    def __init__(self, camera_id, source, db_path, media_dir, segment_sec=60, keyframe_sec=1):
        self.camera_id = camera_id
        self.source = source
        self.db_path = db_path
        self.media_dir = Path(media_dir); self.media_dir.mkdir(parents=True, exist_ok=True)
        self.fps = source.fps()
        self.segment_frames = max(1, int(self.fps * segment_sec))
        self.kf_frames = max(1, int(self.fps * keyframe_sec))

    def run(self):
        writer = None
        seg_idx = 0
        seg_start_n = 0
        seg_start_wall = int(time.time() * 1000)
        pending_frames = []   # (captured_ms, offset_ms, file_path)，段关闭后统一落库
        last_kf = None
        n = 0
        try:
            while True:
                ok, frame = self.source.read()
                if not ok:
                    break
                if writer is None or n - seg_start_n >= self.segment_frames:
                    if writer is not None:
                        writer.release()
                        self._close_segment(seg_idx, seg_start_wall, pending_frames)
                        pending_frames = []; seg_idx += 1
                    seg_start_n = n
                    seg_start_wall = int(time.time() * 1000)
                    writer = self._new_writer(seg_idx, frame)
                writer.write(frame)
                if last_kf is None or n - last_kf >= self.kf_frames:
                    now_ms = int(time.time() * 1000)
                    rel = f"c{self.camera_id}_{seg_idx}_{now_ms}.jpg"
                    cv2.imwrite(str(self.media_dir / rel), frame)
                    pending_frames.append((now_ms, now_ms - seg_start_wall, rel))
                    last_kf = n
                n += 1
        finally:
            if writer is not None:
                writer.release()
                self._close_segment(seg_idx, seg_start_wall, pending_frames)
            self.source.release()

    def _close_segment(self, seg_idx, seg_start_wall, pending_frames):
        end_ms = int(time.time() * 1000)
        seg_id = insert_video_segment(self.db_path, self.camera_id, seg_start_wall, end_ms,
                                      self._seg_path(seg_idx))
        for captured_ms, offset_ms, fpath in pending_frames:
            insert_frame(self.db_path, self.camera_id, seg_id, captured_ms, offset_ms, fpath)

    def _seg_path(self, seg_idx):
        return f"c{self.camera_id}_{seg_idx}.mp4"

    def _new_writer(self, seg_idx, frame):
        h, w = frame.shape[:2]
        return cv2.VideoWriter(str(self.media_dir / self._seg_path(seg_idx)),
                               cv2.VideoWriter_fourcc(*"mp4v"), self.fps, (w, h))
```

```python
"""CLI 入口与录制编排。"""
import sys
import time
import threading
from pathlib import Path
import numpy as np

from config import load_config
from store.db import init_db, insert_camera, insert_transcription, insert_event
from capture.recorder import CameraRecorder, CameraVideoSource, FileVideoSource
from capture.audio import FileAudioSource, MicAudioSource
from perception.asr import Transcriber

PROJECT_ROOT = Path(__file__).resolve().parents[1]

def _resolve(cfg, root, key, default):
    p = Path(cfg["storage"].get(key, default))
    return str(p if p.is_absolute() else root / p)

def run_record(cfg, root, stop_event=None):
    db = _resolve(cfg, root, "db", "data/audiovisual.sqlite")
    media = _resolve(cfg, root, "media_dir", "data/media")
    init_db(db)
    stop = stop_event or threading.Event()

    # 注册摄像头（config id 即 cameras.id，多路一致）
    for c in cfg.get("cameras", []):
        insert_camera(db, c["name"], str(c["source"]), cam_id=c["id"])

    # 每路一个录制线程
    recorders = []
    for c in cfg.get("cameras", []):
        src = c["source"]
        src_obj = (FileVideoSource(src) if isinstance(src, str) and src.endswith((".mp4", ".mkv", ".avi"))
                   else CameraVideoSource(src))
        recorders.append(CameraRecorder(c["id"], src_obj, db, media,
                                        cfg["recording"]["segment_sec"],
                                        cfg["recording"]["keyframe_sec"]))

    def _record_all():
        for r in recorders:
            r.run()
        stop.set()

    threading.Thread(target=_record_all, daemon=True).start()

    # 音频 + 转写线程
    trans = Transcriber(cfg["whisper"]["model"], cfg["whisper"]["device"],
                        cfg["whisper"].get("language", "zh"))
    audio_src = (FileAudioSource(cfg["audio"]["device"], cfg["audio"]["sample_rate"])
                 if cfg["audio"]["source"] == "file" else
                 MicAudioSource(cfg["audio"]["device"], cfg["audio"]["sample_rate"]))
    window_samples = cfg["audio"]["sample_rate"] * cfg["audio"]["window_sec"]

    def _audio_loop():
        buf = np.zeros(0, dtype=np.float32)
        t0 = None
        for chunk in audio_src.chunks():
            if t0 is None:
                t0 = int(time.time() * 1000)
            buf = np.concatenate([buf, chunk])
            while len(buf) >= window_samples:
                win, buf = buf[:window_samples], buf[window_samples:]
                for s in trans.transcribe(win, cfg["audio"]["sample_rate"]):
                    insert_transcription(db, None, t0 + int(s["start"] * 1000),
                                         t0 + int(s["end"] * 1000), s["text"])
                t0 += window_samples * 1000 // cfg["audio"]["sample_rate"]
        stop.set()

    threading.Thread(target=_audio_loop, daemon=True).start()

    try:
        while not stop.is_set():
            time.sleep(0.2)
    except KeyboardInterrupt:
        stop.set()
    insert_event(db, int(time.time() * 1000), "system", "record stopped")

def main(argv=None):
    argv = argv or sys.argv[1:]
    if not argv:
        print("usage: python app.py record|ask|repl|analyze"); return
    cfg = load_config(str(PROJECT_ROOT / "config.yaml"))
    if argv[0] == "record":
        run_record(cfg, PROJECT_ROOT)
    else:
        print(f"'{argv[0]}' 将在 Task 10 实现")

if __name__ == "__main__":
    main()
```

```python
"""检索：FTS5(trigram) 关键词 + 时间/摄像头过滤。"""
from store.db import connect

def search_transcripts(db_path, q, start_ms=None, end_ms=None, camera_id=None, limit=20, desc=False):
    q = (q or "").strip()
    if len(q) >= 3:
        cond = "t.id IN (SELECT rowid FROM transcriptions_fts WHERE transcriptions_fts MATCH ?)"
        params = [f'"{q}"']
    else:
        cond = "t.text LIKE ?"
        params = [f"%{q}%"]
    where = [cond]
    if start_ms is not None:
        where.append("t.start_ms >= ?"); params.append(start_ms)
    if end_ms is not None:
        where.append("t.end_ms <= ?"); params.append(end_ms)
    if camera_id is not None:
        where.append("t.camera_id = ?"); params.append(camera_id)
    order = "DESC" if desc else "ASC"
    sql = f"SELECT t.* FROM transcriptions t WHERE {' AND '.join(where)} ORDER BY t.start_ms {order} LIMIT ?"
    params.append(limit)
    conn = connect(db_path)
    rows = [dict(r) for r in conn.execute(sql, params).fetchall()]
    conn.close()
    return rows

def search_clip_analyses(db_path, q, start_ms=None, end_ms=None, camera_id=None, limit=20, desc=False):
    q = (q or "").strip()
    if len(q) >= 3:
        cond = "c.id IN (SELECT rowid FROM clip_analyses_fts WHERE clip_analyses_fts MATCH ?)"
        params = [f'"{q}"']
    else:
        cond = "c.summary LIKE ?"
        params = [f"%{q}%"]
    where = [cond]
    if start_ms is not None:
        where.append("c.start_ms >= ?"); params.append(start_ms)
    if end_ms is not None:
        where.append("c.end_ms <= ?"); params.append(end_ms)
    if camera_id is not None:
        where.append("c.camera_id = ?"); params.append(camera_id)
    order = "DESC" if desc else "ASC"
    sql = f"SELECT c.* FROM clip_analyses c WHERE {' AND '.join(where)} ORDER BY c.start_ms {order} LIMIT ?"
    params.append(limit)
    conn = connect(db_path)
    rows = [dict(r) for r in conn.execute(sql, params).fetchall()]
    conn.close()
    return rows

def frames_in_range(db_path, camera_id, start_ms, end_ms, limit=50):
    conn = connect(db_path)
    rows = [dict(r) for r in conn.execute(
        "SELECT * FROM frames WHERE camera_id=? AND captured_ms BETWEEN ? AND ? ORDER BY captured_ms LIMIT ?",
        (camera_id, start_ms, end_ms, limit)).fetchall()]
    conn.close()
    return rows
```

```python
"""问答：组装上下文 + 调 Ollama。"""
import requests

def _fmt_ms(ms):
    s = ms // 1000
    return f"{s//3600:02d}:{(s%3600)//60:02d}:{s%60:02d}"

def build_context(transcriptions, clip_summaries=None):
    parts = []
    for t in transcriptions:
        parts.append(f"[{_fmt_ms(t['start_ms'])}] {t['text']}")
    for c in clip_summaries or []:
        parts.append(f"[画面 {_fmt_ms(c['start_ms'])}-{_fmt_ms(c['end_ms'])}] {c['summary']}")
    return "\n".join(parts)

def ask_ollama(host, model, question, context, images=None):
    prompt = (
        "你是视频监控记录助手。只依据下面带时间戳的记录回答，记录没有的就回答「记录里没有」，不要编造。\n\n"
        f"记录：\n{context}\n\n问题：{question}"
    )
    msg = {"role": "user", "content": prompt}
    if images:
        msg["images"] = images  # base64 列表
    r = requests.post(f"{host}/api/chat",
                      json={"model": model, "messages": [msg], "stream": False},
                      timeout=300)
    r.raise_for_status()
    return r.json()["message"]["content"]
```

```python
"""时序理解：视频片段 → 抽多帧 → VLM 动态描述 → clip_analyses。"""
import base64
from pathlib import Path
import cv2
import numpy as np
import requests
from store.db import connect, insert_clip_analysis

def sample_frames(segment_path, start_offset_ms, end_offset_ms, n=8):
    cap = cv2.VideoCapture(str(segment_path))
    fps = cap.get(cv2.CAP_PROP_FPS) or 30
    total = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
    if total <= 0:
        cap.release(); return []
    start_f = min(int(start_offset_ms / 1000 * fps), total - 1)
    end_f = min(int(end_offset_ms / 1000 * fps), total - 1)
    if end_f < start_f:
        end_f = start_f
    indices = np.linspace(start_f, end_f, n, dtype=int)
    frames = []
    for i in indices:
        cap.set(cv2.CAP_PROP_POS_FRAMES, int(i))
        ok, frame = cap.read()
        if ok:
            frames.append(frame)
    cap.release()
    return frames

def _to_b64(frame):
    ok, buf = cv2.imencode(".jpg", frame)
    return base64.b64encode(buf.tobytes()).decode("ascii")

def _segments_in_range(db_path, camera_id, start_ms, end_ms):
    conn = connect(db_path)
    rows = [dict(r) for r in conn.execute(
        "SELECT * FROM video_segments WHERE camera_id=? AND start_ms < ? AND end_ms > ? ORDER BY start_ms",
        (camera_id, end_ms, start_ms)).fetchall()]
    conn.close()
    return rows

def analyze_clip(db_path, media_dir, camera_id, start_ms, end_ms, host, vlm, n=8):
    segs = _segments_in_range(db_path, camera_id, start_ms, end_ms)
    frames = []
    for s in segs:
        path = Path(media_dir) / s["file_path"]
        off_start = max(0, start_ms - s["start_ms"])
        off_end = min(s["end_ms"] - s["start_ms"], end_ms - s["start_ms"])
        per = max(1, n // max(1, len(segs)))
        frames.extend(sample_frames(str(path), off_start, off_end, n=per))
    if not frames:
        return None
    images = [_to_b64(f) for f in frames]
    prompt = "这些是按时间顺序排列的连续帧。请用一两句话描述这段时间里发生了什么（动作/变化/出现的人和物）。"
    r = requests.post(f"{host}/api/chat",
                      json={"model": vlm,
                            "messages": [{"role": "user", "content": prompt, "images": images}],
                            "stream": False},
                      timeout=300)
    r.raise_for_status()
    summary = r.json()["message"]["content"].strip()
    insert_clip_analysis(db_path, camera_id, start_ms, end_ms, summary, vlm)
    return summary
```

```python
def answer_question(cfg, root, question, camera_id=None):
    db = _resolve(cfg, root, "db", "data/audiovisual.sqlite")
    host = cfg["ollama"]["host"]
    now = int(time.time() * 1000)
    window_ms = 10 * 60 * 1000  # 最近 10 分钟
    trans = search_transcripts(db, "", start_ms=now - window_ms, end_ms=now,
                               camera_id=camera_id, limit=30, desc=True)
    clips = search_clip_analyses(db, "", start_ms=now - window_ms, end_ms=now,
                                 camera_id=camera_id, limit=10, desc=True)
    ctx = build_context(trans, clips)
    return ask_ollama(host, cfg["ollama"]["llm"], question, ctx)

def analyze_range(cfg, root, camera_id, start_ms, end_ms):
    db = _resolve(cfg, root, "db", "data/audiovisual.sqlite")
    media = _resolve(cfg, root, "media_dir", "data/media")
    return analyze_clip(db, media, camera_id, start_ms, end_ms,
                        cfg["ollama"]["host"], cfg["ollama"]["vlm"])

def _parse_range(s):
    # CLI 输入便利：HH:MM:SS-HH:MM:SS 按「今天本地墙钟」解析，内部立即转 UTC ms 存储
    a, b = s.split("-")
    def to_ms(hms):
        h, m, sec = (int(x) for x in hms.split(":"))
        today = datetime.date.today()
        dt = datetime.datetime(today.year, today.month, today.day, h, m, sec)
        return int(dt.timestamp() * 1000)
    return to_ms(a), to_ms(b)
```
